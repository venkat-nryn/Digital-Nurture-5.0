Media Object HTML Structure
<!-- Basic Media Object -->
<div class="d-flex gap-3">
    <img src="image.jpg" alt="Media" class="rounded" width="80" height="80">
    <div>
        <h5 class="mt-0">Media Heading</h5>
        <p>Content goes here...</p>
    </div>
</div>

<!-- Nested Media Objects -->
<div class="d-flex gap-3">
    <img src="avatar.jpg" class="rounded-circle" width="60" height="60">
    <div>
        <h6>Parent Comment</h6>
        <p>Parent content...</p>
        <!-- Nested media -->
        <div class="d-flex gap-3 mt-3 ms-4">
            <img src="avatar2.jpg" class="rounded-circle" width="50" height="50">
            <div>
                <h6>Child Comment</h6>
                <p>Child content...</p>
            </div>
        </div>
    </div>
</div>
