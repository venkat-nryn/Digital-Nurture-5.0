Gradient Buttons

How to use bg-gradient
<!-- Basic gradient background -->
<div class="bg-dark bg-gradient text-white">
    Content with dark gradient background
</div>

<!-- Custom gradient with Bootstrap classes -->
<div class="bg-primary bg-gradient text-white">
    Primary gradient background
</div>

<!-- Dark gradient card -->
<div class="card bg-dark bg-gradient text-white">
    <div class="card-body">
        <h5>Gradient Card</h5>
        <p>This card uses dark gradient background</p>
    </div>
</div>
